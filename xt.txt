<script>
    const targetUrl = "LINKTARGET"; // GANTI DENGAN LINK TUJUAN ANDA

    // Elemen DOM
    const elCfState = document.getElementById('cf-state');
    const elAdblock = document.getElementById('adblock-state');
    const elBot = document.getElementById('bot-state');
    const elSuccess = document.getElementById('success-state');
    const elCfSpinner = document.getElementById('cf-spinner');

    // 1. Fungsi Deteksi Bot Dasar (Headless Browser)
    function checkIsBot() {
        if (navigator.webdriver) return true;
        if (window.document.documentElement.getAttribute("webdriver")) return true;
        if (!navigator.language || navigator.languages.length === 0) return true;
        return false;
    }

    // 2. Fungsi Deteksi Adblock
    function checkAdblock() {
        let isBlocked = false;
        
        const adsLoaded = window.adsbygoogle && window.adsbygoogle.loaded === true;
        if (!document.querySelector('script[src*="adsbygoogle"]') || (window.adsbygoogle && !adsLoaded)) {
            isBlocked = true;
        }

        const testAd = document.createElement('div');
        testAd.innerHTML = '&nbsp;';
        testAd.className = 'ad-banner adsbox ad-placement doubleclick';
        document.body.appendChild(testAd);

        if (testAd.offsetHeight === 0 || window.getComputedStyle(testAd).display === 'none') {
            isBlocked = true;
        }
        
        testAd.remove();
        return isBlocked;
    }

    // Cek Bot dasar di awal sebelum Cloudflare dirender
    window.onload = function() {
        if (checkIsBot()) {
            elCfState.classList.add('hidden');
            elBot.classList.remove('hidden');
        }
    };

    // 3. Callback Sukses dari Cloudflare Turnstile
    // Fungsi ini hanya akan berjalan jika pengunjung adalah manusia asli
    function onTurnstileSuccess(token) {
        elCfSpinner.classList.add('hidden'); // Sembunyikan spinner
        
        // Jeda sebentar untuk UX yang bagus, lalu cek Adblock
        setTimeout(function() {
            if (checkAdblock()) {
                // Jika pakai Adblock
                elCfState.classList.add('hidden');
                elAdblock.classList.remove('hidden');
            } else {
                // Jika lulus verifikasi Cloudflare & Adblock dimatikan
                elCfState.classList.add('hidden');
                elSuccess.classList.remove('hidden');
                
                // Redirect ke tujuan
                setTimeout(() => {
                    window.location.href = targetUrl;
                }, 1500);
            }
        }, 1000);
    }
</script>
